\section{Tổng quan vấn đề nghiên cứu và mục đích nghiên cứu}
\subsection{Tổng quan vấn đề nghiên cứu}

\begin{frame}{Bài toán dự báo phụ tải chuỗi thời gian đa biến}
\small
\begin{itemize}
    \item \textbf{Bài toán:} từ $L$ bước quá khứ của $C$ biến, dự báo đồng thời $S$ bước tương lai của cả $C$ biến
          (năng lượng, thời tiết, giao thông\dots).

    \vspace{0.1cm}
    \item \textbf{Vai trò:} dự báo phụ tải được xem là năng lực nền tảng cho quy hoạch và vận hành hệ thống
          điện\refnote{Hong2016Probabilistic}.

    \vspace{0.1cm}
    \item \textbf{Thách thức đa biến:} nắm phụ thuộc giữa các kênh và xử lý dịch chuyển phân phối ngày càng khó khi quy mô
          và lĩnh vực ứng dụng mở rộng\refnote{Kim2025Survey}.

    \vspace{0.1cm}
    \item \textbf{Xu hướng hiện nay:} dùng cửa sổ quá khứ rất dài ($L = 720$) để nắm phụ thuộc dài hạn, kéo theo số
          tham số và khối lượng tính toán tăng mạnh\refnote{Huang2025TimeBase}.

    \vspace{0.1cm}
    \item \textbf{Ba yêu cầu cần đáp ứng cùng lúc:}
    \begin{itemize}
        \item \emph{Nhẹ và nhanh}, để triển khai được trên thiết bị hạn chế tài nguyên.
        \item \emph{Khai thác mẫu hình chu kỳ} của chuỗi.
        \item \emph{Học quan hệ giữa các biến} mà không khuếch đại các phụ thuộc giả.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Đặc điểm 1: Tính lặp lại}
\begin{itemize}
    \item \textbf{Tương đồng theo chu kỳ:} các đoạn dài một chu kỳ (ví dụ mỗi ngày) có hình dạng rất giống nhau, với hệ số
          tương quan Pearson giữa các đoạn cao\refnote{Huang2025TimeBase}.

    \vspace{0.1cm}
    \item \textbf{Phụ thuộc theo chu kỳ toàn cục:} tương quan giữa một đoạn và đoạn tương ứng ở xa trong chu kỳ toàn cục
          thường cao hơn tương quan với các đoạn liền kề\refnote{Cao2026GTR}; dữ liệu điện có mẫu hình ngày dùng chung cho
          mọi ngày\refnote{Lin2024CycleNet}.

    \vspace{0.1cm}
    \item \textbf{Hệ quả:} không nhất thiết phải mô hình hóa từng điểm thời gian. TimeBase chỉ dùng
          \textbf{0.39K tham số}, ít hơn DLinear hơn $2.6\times10^{3}$ lần về tham số và 120 lần về MACs, mà vẫn đạt độ chính
          xác cạnh tranh.
\end{itemize}

\vspace{0.1cm}
\begin{block}{}
\centering Dự báo trên \textbf{một vài ``đoạn đại diện''} thay vì trên $L$ điểm thời gian.
\end{block}
\end{frame}

\begin{frame}{Đặc điểm 2: Phụ thuộc chéo giữa các biến}
\footnotesize
\begin{itemize}
    \item \textbf{Hai nhóm phương pháp theo cách xử lý biến}\refnote{Han2024Capacity}:
    \begin{itemize}\footnotesize
        \item \textbf{Độc lập biến (CI)} (PatchTST, DLinear, TimeBase): bền vững nhưng bỏ qua tương tác giữa các kênh.
        \item \textbf{Phụ thuộc biến (CD)} (iTransformer, DyTimeNet): nắm được tương tác nhưng dễ bị nhiễu từ các kênh không
              liên quan\refnote{Qiu2025DUET}.
    \end{itemize}

    \item \textbf{Phụ thuộc thay đổi theo trạng thái:} cùng một cặp biến có thể liên kết mạnh ở kịch bản này nhưng yếu ở
          kịch bản khác; nhiễu cục bộ tạo ra tương quan giả\refnote{Zhang2026MSFLOW}.

    \item \textbf{Mẫu khác toàn cục:} tương quan đo trên từng mẫu có thể lệch đáng kể so với tương quan toàn cục của tập
          huấn luyện\refnote{Lin2025TQNet}.

    \item \textbf{Phụ thuộc thưa:} DyTimeNet dùng chiến lược thưa \emph{DropFusion} để giảm thông tin dư thừa khi học quan
          hệ liên biến\refnote{Chen2026DyTimeNet}.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Sparse segment}
\footnotesize
\begin{itemize}
    \item \textbf{Công trình tiêu biểu:} PatchTST chia mỗi chuỗi đơn biến thành các patch làm token cho
          Transformer\refnote{Nie2023PatchTST}; TimesNet biến chuỗi 1D thành tensor 2D theo các chu kỳ để nắm biến thiên trong
          và giữa các chu kỳ\refnote{Wu2023TimesNet}; SparseTSF dự báo tuyến tính giữa các chu
          kỳ\refnote{Lin2024SparseTSF}; TimeBase dự báo trên một số ít đoạn cơ sở\refnote{Huang2025TimeBase}.
    \item \textbf{So sánh, ưu điểm:} PatchTST dùng patch độ dài cố định, TimesNet tổ chức chuỗi theo chu kỳ của dữ liệu;
          token mức đoạn bền vững hơn với nhiễu cục bộ và rút ngắn chuỗi cần xử lý.
    \item \textbf{Nhược điểm:} đoạn được đưa vào backbone sâu nên mô hình vẫn phức tạp; SparseTSF, TimeBase rất nhẹ nhưng xử lý
          từng biến độc lập.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Multivariate time series}
\small
\begin{itemize}
    \item \textbf{Công trình tiêu biểu:} Informer cải thiện hiệu quả attention trên chuỗi dài\refnote{Zhou2021Informer};
          FEDformer tích hợp phân rã với cấu trúc miền tần số\refnote{Zhou2022FEDformer}; JTFT kết hợp miền thời gian và
          miền tần số, dùng attention hạng thấp để nắm phụ thuộc giữa các chiều\refnote{Chen2024JTFT}.
    \item \textbf{So sánh:} khác nhau ở cách biểu diễn tín hiệu (điểm, token tần số, thời gian--tần số), nhưng cùng dựa
          trên mạng sâu.
    \item \textbf{Ưu điểm:} biểu diễn mạnh đối với động học phức tạp, phi tuyến.
    \item \textbf{Nhược điểm}: chi phí backbone sâu lớn hơn nhiều so với ánh xạ tuyến tính; quan
          hệ liên biến được học lẫn trong backbone thay vì qua một cơ chế thưa, tường minh.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Load forecasting}
\small
\begin{itemize}
    \item \textbf{Mô hình thống kê:} ARIMA đa mùa vụ thích nghi dự báo phụ tải nước nóng sinh hoạt\refnote{Nagpal2025ARIMA};
          SARIMA kết hợp phân tích phổ dự báo nhu cầu điện đỉnh theo mùa\refnote{MelchorCruz2025Spectral}; các cách cổ
          điển (Fourier, làm trơn mũ, SARIMA) chưa phản ánh đặc trưng riêng và tương tác giữa các chu kỳ mùa
          vụ\refnote{Moon2022MixedPooling}.
    \item \textbf{Mô hình học sâu:} DeepAR học một mô hình tự hồi quy toàn cục trên nhiều chuỗi liên quan và đưa ra phân
          phối dự báo\refnote{Salinas2020DeepAR}; TFT kết hợp chọn biến với attention có thể diễn giải cho dự báo đa
          tầm\refnote{Lim2021TFT}.
    \item \textbf{So sánh:} mô hình thống kê minh bạch, rẻ nhưng dựa trên giả định tuyến tính và thường xử lý các thành phần
          mùa vụ riêng rẽ; mô hình sâu nắm được tính phi tuyến nhưng nặng và không mô hình hóa tường minh việc biến nào ảnh
          hưởng biến nào.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Dynamic dependency (1/2)}
\small
\textbf{Công trình tiêu biểu:}
\begin{itemize}
    \item \textbf{MS-FLOW:} gộp token patch thành trạng thái mức biến, chỉ giữ Top-$K$ cạnh cho mỗi biến trong đồ thị thích
          nghi theo mẫu, rồi lan truyền thông tin dọc các cạnh này\refnote{Zhang2026MSFLOW}.
    \item \textbf{xCPD:} mỗi cặp kênh--patch là một nút đồ thị, chia thành dải tần thấp/trung/cao, định tuyến tới các
          chuyên gia theo tần số\refnote{Li2026xCPD}.
    \item \textbf{DUET:} phân cụm mềm các kênh trong miền tần số bằng metric Mahalanobis học được và mặt nạ lấy mẫu
          Bernoulli, hợp nhất qua masked attention\refnote{Qiu2025DUET}.
    \item \textbf{TQNet:} query học được dịch theo chu kỳ, key/value từ đầu vào gốc, chỉ một lớp attention kết hợp tương
          quan toàn cục và tương quan mức mẫu\refnote{Lin2025TQNet}.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Dynamic dependency (2/2)}
\small
\begin{itemize}
    \item \textbf{Ưu điểm:} mô hình hóa phụ thuộc bền vững; TQNet và xCPD còn chú trọng hiệu quả.
    \item \textbf{Nhược điểm:} học trên token mức patch/điểm nên tăng theo $C$: DUET dùng ma trận $C\times C$ giữa mọi cặp
          kênh, xCPD dựng đồ thị $C\times N$ nút; MS-FLOW phải chọn $K$ theo từng bộ dữ liệu; TQNet: ép mô hình hóa đa biến
          có thể gây hại khi tương quan yếu.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Pattern extraction (1/2)}
\small
\textbf{Công trình tiêu biểu:}
\begin{itemize}
    \item \textbf{CycleNet:} học các chu kỳ lặp $Q\in\R^{W\times C}$ dùng chung cho cả bộ dữ liệu, loại khỏi đầu vào và chỉ
          dự báo phần dư; chỉ thêm $W\times C$ tham số, không thêm MACs\refnote{Lin2024CycleNet}.
    \item \textbf{TimeEmb:} tách thành phần bất biến (truy xuất từ ngân hàng embedding theo dấu thời gian) khỏi thành phần
          biến đổi (xử lý bằng bộ lọc tần số)\refnote{Xia2025TimeEmb}.
    \item \textbf{GTR:} embedding toàn cục cho cả một chu kỳ, truy xuất đoạn theo vị trí tuyệt đối, hợp nhất bằng tích chập
          2D; mô-đun chỉ 40.1K tham số\refnote{Cao2026GTR}.
    \item \textbf{LightGTS:} patch dài đúng một chu kỳ nội tại, giải mã song song; nhỏ hơn 10--100 lần các mô hình nền tảng
          khác\refnote{Wang2025LightGTS}.
\end{itemize}
\end{frame}

\begin{frame}{Related works: Pattern extraction (2/2)}
\small
\begin{itemize}
    \item \textbf{So sánh:} CycleNet, TimeEmb, GTR lưu mẫu hình \emph{toàn cục} học trên cả tập huấn luyện; LightGTS trích
          mẫu hình \emph{cục bộ} căn theo chu kỳ từ cửa sổ đầu vào.
    \item \textbf{Ưu điểm:} nắm cấu trúc chu kỳ với chi phí thấp.
    \item \textbf{Nhược điểm:} mẫu hình được trích xuất theo từng kênh. CycleNet thua iTransformer trên Traffic và nêu mô
          hình hóa liên kênh là hướng phát triển tương lai\refnote{Lin2024CycleNet}.
\end{itemize}
\end{frame}

\subsection{Mục đích nghiên cứu}

\begin{frame}{Mục đích nghiên cứu}
\small
Đề xuất mô hình \textbf{DyBAS-Net} (\textbf{Dy}namic cross-variable dependency on \textbf{BAS}is components), với các mục
tiêu:
\begin{itemize}
    \item \textbf{Trích xuất mẫu hình dựa trên đoạn thưa:} phân đoạn theo chu kỳ, chuẩn hóa từng pha, nén $N$ đoạn quá khứ
          thành $R\ll N$ đoạn cơ sở trực giao.

    \item \textbf{Học phụ thuộc động trong không gian cơ sở:} đưa CrossVBlock với chiến lược DropFusion và IntraVBlock của
          DyTimeNet\refnote{Chen2026DyTimeNet} vào xử lý $P\cdot R$ hệ số cơ sở của mỗi biến, thay vì $L$ điểm thời gian.

    \item \textbf{Hiệu chỉnh an toàn:} nhánh trộn biến chỉ cộng một lượng hiệu chỉnh vào các thành phần cơ sở, khởi tạo bằng
          0, nên mô hình xuất phát đúng từ backbone CI.

    \item \textbf{Hiệu quả:} phép tính trên trục thời gian chỉ ở kích thước mức đoạn ($N$, $R$, $P$, cỡ hàng chục) thay vì
          $L$, $S$ (cỡ hàng trăm).

    \item \textbf{Đánh giá toàn diện:} so sánh độ chính xác với 14 mô hình trên 5 bộ dữ liệu, và đo chi phí tính toán của
          các mô hình trên \emph{cùng một thiết bị}.
\end{itemize}
\end{frame}

\begin{frame}{Input}
    \small
    \textbf{Đầu vào $\mathcal{X} \in \mathbb{R}^{L \times C}$:} một cửa sổ quá khứ độ dài $L$ của $C$ biến,
    $i = \overline{1, C}$ (cửa sổ được trượt trên cả tập train, val và test).

    \begin{columns}[c]
        \begin{column}{0.52\textwidth}
            \[
            \mathcal{X} =
            \begin{bmatrix}
            x_1^{(1)} & x_1^{(2)} & \dots & x_1^{(C)} \\
            x_2^{(1)} & x_2^{(2)} & \dots & x_2^{(C)} \\
            \vdots & \vdots & \ddots & \vdots \\
            x_L^{(1)} & x_L^{(2)} & \dots & x_L^{(C)}
            \end{bmatrix} \tag{\ref*{P-eq:input}}\]
        \end{column}
        \begin{column}{0.45\textwidth}
            \textbf{Định nghĩa thành phần:}
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item $x_t^{(i)}$: Giá trị của biến thứ $i$ tại thời điểm $t$.
                \item $i = \overline{1, C}$: Chỉ số biến.
                \item $t = \overline{1, L}$: Chỉ số mốc thời gian.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.1cm}
    \textbf{Giá trị thực $\mathcal{Y} \in \mathbb{R}^{S \times C}$:} giá trị của $S$ bước ngay sau cửa sổ:
    \[
    \mathcal{Y} =
    \begin{bmatrix}
    x_{L+1}^{(1)} & x_{L+1}^{(2)} & \dots & x_{L+1}^{(C)} \\
    \vdots & \vdots & \ddots & \vdots \\
    x_{L+S}^{(1)} & x_{L+S}^{(2)} & \dots & x_{L+S}^{(C)}
    \end{bmatrix} \tag{\ref*{P-eq:label}}\]
\end{frame}

\begin{frame}{Output và Hàm mục tiêu}
    \begin{itemize}
        \item \textbf{Ma trận kết quả dự báo ($\hat{\mathcal{Y}} \in \mathbb{R}^{S \times C}$):} được ước lượng đồng thời
              bởi mô hình $\mathcal{F}$ (\textbf{DyBAS-Net}):
        $$\hat{\mathcal{Y}} = \mathcal{F}(\mathcal{X})$$

        \item \textbf{Sai số dự báo (MSE):}
        \[ \mathcal{L}_{MSE} = \frac{1}{S \times C} \sum_{t=1}^{S} \sum_{i=1}^{C} \left( x_{L+t}^{(i)} - \hat{x}_{L+t}^{(i)} \right)^2 \tag{\ref*{P-eq:mse}}\]

        \item \textbf{Hàm mục tiêu khi huấn luyện:} thêm ràng buộc trực giao trên các thành phần cơ sở:
        \[\mathcal{L} = \mathcal{L}_{MSE} + \lambda_{orth}\,\mathcal{L}_{orth}, \qquad \lambda_{orth}\ge 0 \tag{\ref*{P-eq:objective}}\]
    \end{itemize}
\end{frame}

\subsection{Xây dựng mô hình}

\begin{frame}{Tổng quan các bước của DyBAS-Net}
\begin{enumerate}
    \item \textbf{Phân đoạn theo chu kỳ:} cắt chuỗi dài $L$ thành $N$ đoạn dài $P$.
    \item \textbf{Chuẩn hóa theo pha:} trừ trung bình của từng pha trong chu kỳ.
    \item \textbf{Trích xuất cơ sở:} nén $N$ đoạn thành $R$ thành phần cơ sở $\mathbf{E}^{(i)}$.
    \item \textbf{Mã hóa hệ số cơ sở} của mỗi biến thành vector $D$ chiều.
    \item \textbf{CrossVBlock + DropFusion:} trộn thông tin giữa các biến.
    \item \textbf{IntraVBlock:} biến đổi phi tuyến trong từng biến.
    \item \textbf{Chiếu ngược:} sinh lượng hiệu chỉnh $\Delta\mathbf{E}^{(i)}$, cộng vào cơ sở.
    \item \textbf{Dự báo theo đoạn và khôi phục:} sinh $N'$ đoạn tương lai, cộng lại trung bình theo pha.
\end{enumerate}

\end{frame}

\begin{frame}{1. Phân đoạn theo chu kỳ}
\small
Chuỗi của biến $i$, $\mathbf{x}^{(i)} = (x^{(i)}_1,\dots,x^{(i)}_L)$, được xếp thành \textbf{ma trận đoạn}
$\mathbf{X}^{(i)}\in\R^{P\times N}$, mỗi cột là một đoạn dài $P$:
\[
\mathbf{X}^{(i)} =
\begin{bmatrix}
x^{(i)}_1 & x^{(i)}_{P+1} & \dots & x^{(i)}_{(N-1)P+1}\\
x^{(i)}_2 & x^{(i)}_{P+2} & \dots & x^{(i)}_{(N-1)P+2}\\
\vdots & \vdots & \ddots & \vdots\\
x^{(i)}_P & x^{(i)}_{2P} & \dots & x^{(i)}_{NP}
\end{bmatrix},
\qquad \big[\mathbf{X}^{(i)}\big]_{p,n} = x^{(i)}_{(n-1)P+p} \tag{\ref*{P-eq:segment}}\]
\begin{itemize}
    \item Ví dụ ETTh1: $L=720$, $P=24$ $\Rightarrow$ $N=30$ cột, mỗi cột là một ngày.
    \item $\big[\mathbf{X}^{(i)}\big]_{p,n}$: phần tử ở hàng $p$, cột $n$ là giá trị ở giờ thứ $p$ của ngày thứ $n$.
          Trước ngày thứ $n$ có $(n-1)$ ngày đầy đủ, tức $(n-1)P$ điểm; đi thêm $p$ bước vào ngày $n$ là tới vị trí
          $(n-1)P+p$ trong chuỗi gốc.
\end{itemize}
\end{frame}


\begin{frame}{2. Chuẩn hóa theo pha}
\begin{itemize}
    \item \textbf{Trung bình theo pha:} trung bình của từng hàng (từng pha) qua $N$ đoạn:
    \[\mu^{(i)}_p = \frac{1}{N}\sum_{n=1}^{N}\big[\mathbf{X}^{(i)}\big]_{p,n},
      \qquad \bm{\mu}^{(i)} = (\mu^{(i)}_1,\dots,\mu^{(i)}_P)^{\top}\in\R^{P} \tag{\ref*{P-eq:phase-mean}}\]

    \item \textbf{Trừ trung bình:}
    \[\big[\tilde{\mathbf{X}}^{(i)}\big]_{p,n} = \big[\mathbf{X}^{(i)}\big]_{p,n} - \mu^{(i)}_p
      \qquad (\text{trừ } \mu^{(i)}_p \text{ cho mọi phần tử của hàng } p) \tag{\ref*{P-eq:period-norm}}\]

    \item \textbf{Ý nghĩa:} $\bm{\mu}^{(i)}$ là ``đoạn trung bình'' của biến $i$.
          Các bước sau chỉ học phần dao động quanh nó.
\end{itemize}
\end{frame}

\begin{frame}{3. Trích xuất thành phần cơ sở (Basis Extraction)}
\begin{itemize}
    \item Ánh xạ tuyến tính \textbf{dùng chung cho mọi biến} nén $N$ đoạn thành $R$ thành phần cơ sở
          ($\mathbf{W}_B\in\R^{N\times R}$, $\mathbf{b}_B\in\R^{R}$ học được):
    \[\mathbf{E}^{(i)} = \tilde{\mathbf{X}}^{(i)}\,\mathbf{W}_B + \mathbf{b}_B \quad\in\R^{P\times R}
      \qquad (\mathbf{b}_B \text{ cộng vào mọi hàng}) \tag{\ref*{P-eq:basis}}\]

    \item \textbf{Cột thứ $r$ của $\mathbf{E}^{(i)}$} là tổ hợp tuyến tính của $N$ đoạn quá khứ
          $\tilde{\mathbf{x}}^{(i)}_n$, tức một ``đoạn đại diện'':
    \[\mathbf{e}^{(i)}_r = \sum_{n=1}^{N} [\mathbf{W}_B]_{n,r}\;\tilde{\mathbf{x}}^{(i)}_{n} + [\mathbf{b}_B]_r
      \quad\in\R^{P} \tag{\ref*{P-eq:basis-col}}\]

    \item \textbf{Ý nghĩa:} tóm $N=30$ ngày thành $R=6$ ``ngày điển hình'', dựa trên tính hạng thấp của chuỗi
          dài\refnote{Huang2025TimeBase}.
\end{itemize}
\end{frame}

\begin{frame}{3. Ràng buộc trực giao trên các thành phần cơ sở}
\begin{itemize}
    \item \textbf{Ma trận Gram:} tích vô hướng giữa từng cặp thành phần cơ sở
    \[\mathbf{G}^{(i)} = \mathbf{E}^{(i)\top}\mathbf{E}^{(i)}\in\R^{R\times R},
      \qquad \big[\mathbf{G}^{(i)}\big]_{r,s} = \langle \mathbf{e}^{(i)}_r,\mathbf{e}^{(i)}_s\rangle \tag{\ref*{P-eq:gram}}\]

    \item \textbf{Phạt phần ngoài đường chéo} (trung bình trên $B$ mẫu của một lô và $C$ biến):
    \[\mathcal{L}_{orth} = \frac{1}{B\,C}\sum_{b=1}^{B}\sum_{i=1}^{C}
      \Big\|\mathbf{G}^{(i)}_b - \operatorname{diag}\big(\mathbf{G}^{(i)}_b\big)\Big\|_F \tag{\ref*{P-eq:orth}}\]

    \item $\mathcal{L}_{orth}=0$ khi các thành phần cơ sở đôi một trực giao, nên mỗi thành phần mang một kiểu biến thiên
          riêng, không trùng lặp.
    \item Chỉ dùng khi huấn luyện; khi đánh giá chỉ đo MSE và MAE.
\end{itemize}

\end{frame}

\begin{frame}{4. Mã hóa hệ số cơ sở (Basis Encoding)}
\begin{itemize}
    \item \textbf{Làm phẳng:} trải $\mathbf{E}^{(i)}$ theo hàng thành vector, rồi ghép $C$ biến thành ma trận:
    \[\mathbf{e}^{(i)} = \operatorname{vec}\big(\mathbf{E}^{(i)}\big)\in\R^{PR},
      \qquad \bar{\mathbf{E}} = \big[\mathbf{e}^{(1)},\dots,\mathbf{e}^{(C)}\big]\in\R^{PR\times C} \tag{\ref*{P-eq:flatten}}\]

    \item \textbf{Mã hóa:} ánh xạ tuyến tính sang không gian ẩn $D$ chiều:
    \[\mathcal{X}_{enc} = \mathbf{W}_{enc}^{\top}\,\bar{\mathbf{E}} + \mathbf{b}_{enc} \quad\in\R^{D\times C}
      \qquad (\mathbf{b}_{enc} \text{ cộng vào mọi cột}) \tag{\ref*{P-eq:embed}}\]
    \begin{itemize}
        \item $\mathbf{W}_{enc}\in\R^{PR\times D}$: ma trận trọng số có thể học; $\mathbf{b}_{enc}\in\R^{D}$: vector độ lệch.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Chuẩn hóa LayerNorm và hàm kích hoạt GELU}
\begin{itemize}
    \item \textbf{LayerNorm}\refnote{Ba2016LayerNorm}: chuẩn hóa $D$ đặc trưng của từng biến (cột $\mathbf{h}\in\R^{D}$),
          đưa mọi biến về cùng thang đo trước khi trộn:
    \[\operatorname{LN}(\mathbf{h}) = \bm{\gamma}\odot\frac{\mathbf{h}-\mu_{\mathbf{h}}}{\sqrt{\sigma^2_{\mathbf{h}}+\varepsilon}}
      + \bm{\beta} \tag{\ref*{P-eq:ln-def}}\]
    \begin{itemize}
        \item $\mu_{\mathbf{h}}, \sigma^2_{\mathbf{h}}$: trung bình, phương sai các phần tử của $\mathbf{h}$;
              $\bm{\gamma},\bm{\beta}\in\R^{D}$ học được.
    \end{itemize}

    \item \textbf{Hàm kích hoạt GELU}\refnote{Hendrycks2016GELU}:
    \[\text{GELU}(x) = x \cdot \Phi(x) = \frac{x}{2}\left(1 + \text{erf}\!\left(\frac{x}{\sqrt{2}}\right)\right) \tag{\ref*{P-eq:gelu}}\]
    \begin{itemize}
        \item $\Phi(x)$: Hàm phân phối tích lũy của $\mathcal{N}(0,1)$;
              $\text{erf}(x) = \frac{2}{\sqrt{\pi}}\int_0^x e^{-t^2}\,dt$.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{5. Khối Liên biến (CrossVBlock)}
\begin{itemize}
    \item \textbf{Chuẩn hóa:}
    $$\tilde{\mathcal{X}}_{enc} = \operatorname{LN}(\mathcal{X}_{enc}) \quad \in \R^{D \times C}$$

    \item \textbf{MLP nút thắt trên trục biến} $C\to C'\to C$ (khi kiểm thử), với
          $\mathbf{W}_{df,1}\in\R^{C\times C'}$, $\mathbf{W}_{df,2}\in\R^{C'\times C}$ (độ lệch $\mathbf{b}_1$, $\mathbf{b}_2$ cộng vào mọi hàng):
    \begin{align*}
      \mathbf{H} &= \text{GELU}\big(\tilde{\mathcal{X}}_{enc}\,\mathbf{W}_{df,1} + \mathbf{b}_{1}\big)
                    &&\in\R^{D\times C'} \tag{\ref*{P-eq:cross1}}
    \\
      \mathcal{X}_{cross} &= \mathbf{H}\,\mathbf{W}_{df,2} + \mathbf{b}_{2} &&\in\R^{D\times C} \tag{\ref*{P-eq:cross2}}
    \end{align*}

    \item \textbf{Vì sao là trộn liên biến:} mỗi \emph{hàng} của $\tilde{\mathcal{X}}_{enc}$ là một đặc trưng của cả $C$ biến.
          Nhân phải với $\mathbf{W}_{df}$ tạo mỗi biến mới bằng tổ hợp của mọi biến trên từng đặc trưng.

    \item Khi huấn luyện, hai lớp tuyến tính này được thay bằng lớp \textbf{DropFusion}\refnote{Chen2026DyTimeNet}.
\end{itemize}
\end{frame}

\begin{frame}{5a. DropFusion: Mặt nạ Cạnh $\mathcal{M}_1$ và Mặt nạ Nút $\mathcal{M}_2$}
    Khi huấn luyện, hai lớp tuyến tính của CrossVBlock được can thiệp bởi DropFusion. Xét lớp thứ nhất
    (đầu vào $\tilde{\mathcal{X}}_{enc}\in\R^{D\times C}$, trọng số $\mathbf{W}_{df,1}\in\R^{C\times C'}$):
    \vspace{0.15cm}
    \begin{itemize}
        \item \textbf{Giai đoạn 1 -- Mặt nạ Cạnh $\mathcal{M}_1\in\{0,1\}^{C\times C'}$:} mỗi phần tử bằng 0 với xác
              suất $p_1$, tức tắt liên kết giữa một biến đầu vào và một nút ẩn. Lấy mẫu lại ở mỗi lô dữ liệu.
        \[\mathbf{W}_{masked} = \mathbf{W}_{df,1} \odot \mathcal{M}_1 \tag{\ref*{P-eq:wmasked}}\]

        \item \textbf{Giai đoạn 2 -- Mặt nạ Nút $\mathcal{M}_2\in\{0,1\}^{D\times C'}$:} mỗi phần tử bằng 0 với xác suất
              $p_2$, tức tắt ngẫu nhiên từng phần tử đầu ra. Lấy mẫu riêng cho từng mẫu.
        \[\mathbf{Y}_{masked} = \big(\tilde{\mathcal{X}}_{enc}\,\mathbf{W}_{masked} + \mathbf{b}_1\big) \odot \mathcal{M}_2 \tag{\ref*{P-eq:ymasked}}\]
    \end{itemize}
    \small Lớp thứ hai làm tương tự, với đầu vào $\mathbf{H}$ và trọng số $\mathbf{W}_{df,2}$.
\end{frame}


\begin{frame}{5b. DropFusion: Chuẩn hóa kỳ vọng}
    \begin{itemize}
        \item \textbf{Vấn đề năng lượng tín hiệu:} khi huấn luyện, một phần kết nối bị tắt ($p_1, p_2$); khi kiểm thử mạng mở
              lại $100\%$ kết nối nên tổng tín hiệu tăng vọt, gây sai lệch.
        \item \textbf{Giải pháp:} nhân hệ số bù để kỳ vọng tín hiệu không đổi giữa hai giai đoạn.
    \end{itemize}

    \begin{alertblock}{Công thức DropFusion hoàn chỉnh (khi huấn luyện)}
        \[\text{DF}(\tilde{\mathcal{X}}_{enc}) = \frac{1}{1-p_2}\left[\frac{\tilde{\mathcal{X}}_{enc}\,(\mathbf{W}_{df,1}\odot\mathcal{M}_1)}{1-p_1}
          + \mathbf{b}_1\right]\odot\mathcal{M}_2 \tag{\ref*{P-eq:dropfusion}}\]
    \end{alertblock}

    % \begin{itemize}
    %     \item \textbf{Khi kiểm thử:} $\mathcal{M}_1=\mathcal{M}_2=\one$, không nhân hệ số bù:
    %           $\text{DF}(\tilde{\mathcal{X}}_{enc}) = \tilde{\mathcal{X}}_{enc}\,\mathbf{W}_{df,1} + \mathbf{b}_1$
    %           (đúng bằng lớp tuyến tính ở CrossVBlock).
    %     \item Lớp thứ hai: thay $\tilde{\mathcal{X}}_{enc}$ bằng $\mathbf{H}$, $\mathbf{W}_{df,1}$ bằng $\mathbf{W}_{df,2}$,
    %           $\mathbf{b}_1$ bằng $\mathbf{b}_2$.
    % \end{itemize}
\end{frame}

\begin{frame}{6. Khối Nội biến (IntraVBlock)}
    \begin{itemize}
        \item \textbf{Kết nối thặng dư:} cộng đặc trưng chéo với đặc trưng đã chuẩn hóa để giữ thông tin gốc của từng biến:
        \[\mathcal{X}_{intra} = \mathcal{X}_{cross} + \tilde{\mathcal{X}}_{enc} \quad \in \R^{D \times C} \tag{\ref*{P-eq:cross-res}}\]

        \item \textbf{Mạng truyền thẳng dùng chung}, áp dụng cho \emph{từng cột} (từng biến) $\mathbf{h}\in\R^{D}$:
        \[\text{FFN}(\mathbf{h}) = \mathbf{W}_4\,\text{GELU}(\mathbf{W}_3\mathbf{h} + \mathbf{b}_3) + \mathbf{b}_4,
          \qquad \mathbf{W}_3\in\R^{D_{ff}\times D},\ \mathbf{W}_4\in\R^{D\times D_{ff}} \tag{\ref*{P-eq:ffn}}\]

        \item \textbf{Đầu ra} (LayerNorm phía trước, thặng dư phía sau):
        \[\hat{\mathcal{X}}_{intra} = \mathcal{X}_{intra} + \text{FFN}\big(\operatorname{LN}(\mathcal{X}_{intra})\big)
          \quad \in \R^{D \times C} \tag{\ref*{P-eq:intra}}\]

    \end{itemize}
\end{frame}

\begin{frame}{7. Chiếu ngược và Hiệu chỉnh thành phần cơ sở}
    \begin{itemize}
        \item \textbf{Chiếu ngược} về không gian cơ sở:
        \[\Delta\bar{\mathbf{E}} = \mathbf{W}_{out}^{\top}\,\hat{\mathcal{X}}_{intra} + \mathbf{b}_{out}
          \quad\in\R^{PR\times C}, \qquad \mathbf{W}_{out}\in\R^{D\times PR} \tag{\ref*{P-eq:proj}}\]

        \item Cột $i$ được xếp lại thành $\Delta\mathbf{E}^{(i)}\in\R^{P\times R}$ và \textbf{cộng vào cơ sở}:
        \[\mathbf{E}'^{(i)} = \mathbf{E}^{(i)} + \Delta\mathbf{E}^{(i)} \tag{\ref*{P-eq:correct}}\]

        \item \textbf{Khởi tạo bằng 0:} $\mathbf{W}_{out}=\mathbf{0}$, $\mathbf{b}_{out}=\mathbf{0}$ nên lúc bắt đầu huấn
              luyện $\Delta\mathbf{E}^{(i)}=\mathbf{0}$.
        \item \textbf{Ý nghĩa:} nhánh trộn biến chỉ học \emph{phần hiệu chỉnh}: dùng thông tin của các biến khác để sửa các
              ``ngày điển hình'' của biến $i$.
    \end{itemize}
\end{frame}

\begin{frame}{8. Dự báo theo đoạn và Khôi phục thang đo}
    \begin{itemize}
        \item \textbf{Dự báo theo đoạn:} ánh xạ tuyến tính dùng chung sinh $N'$ đoạn tương lai từ $R$ cơ sở đã hiệu chỉnh:
        \[\hat{\mathbf{Y}}^{(i)} = \mathbf{E}'^{(i)}\,\mathbf{W}_F + \mathbf{b}_F \quad\in\R^{P\times N'},
          \qquad \mathbf{W}_F\in\R^{R\times N'} \tag{\ref*{P-eq:forecast}}\]

        \item \textbf{Khôi phục thang đo và ghép đoạn:} cộng trả trung bình theo pha cho mọi đoạn tương lai, nối các
              cột thành chuỗi và giữ $S$ bước đầu:
        \[\hat{x}^{(i)}_{L+t} = \big[\hat{\mathbf{Y}}^{(i)}\big]_{p,n} + \mu^{(i)}_p,\quad t=(n-1)P+p,\ \ t=\overline{1,S} \tag{\ref*{P-eq:merge}}\]

        \item Ghép $C$ biến cho dự báo $\hat{\mathcal{Y}}\in\R^{S\times C}$.
    \end{itemize}
\end{frame}

\begin{frame}{Điều chỉnh và Tối ưu hóa Trọng số}
    \begin{itemize}
        \item Trọng số được cập nhật ngược hướng gradient của hàm mục tiêu $\mathcal{L}$:
        \[ \bm{\theta}_{k+1} = \bm{\theta}_k - \alpha\,\frac{\partial \mathcal{L}}{\partial \bm{\theta}_k} \tag{\ref*{P-eq:gd}}\]
        với $\bm{\theta}$ gồm mọi trọng số $\mathbf{W}$ và độ lệch $\mathbf{b}$, $k$ là bước cập nhật.

        \item \textbf{Hai tốc độ học:} $\alpha=\alpha_B$ cho $\mathbf{W}_B,\mathbf{b}_B,\mathbf{W}_F,\mathbf{b}_F$,
              $\alpha=\alpha_M$ cho nhánh trộn biến.

        \item \textbf{Tốc độ học} theo epoch $\tau$: giữ nguyên 4 epoch đầu, sau đó nhân $0.8$ mỗi epoch:
        $\alpha^{(\tau)} = \alpha$ nếu $\tau\le 4$, \ $\alpha^{(\tau)} = \alpha\cdot 0.8^{\,\tau-4}$ nếu $\tau>4$ \hfill(\ref*{P-eq:lr})
    \end{itemize}
\end{frame}

\begin{frame}{Chỉ số đánh giá}
\begin{itemize}
    \item \textbf{Độ chính xác:} MSE và MAE, lấy trung bình trên $M$ cửa sổ của tập test ($m$: cửa sổ thứ $m$):
    \begin{align*}
      \text{MSE} &= \frac{1}{MSC}\sum_{m=1}^{M}\sum_{t=1}^{S}\sum_{i=1}^{C}\big(x^{(i)}_{m,L+t}-\hat{x}^{(i)}_{m,L+t}\big)^2\\
      \text{MAE} &= \frac{1}{MSC}\sum_{m=1}^{M}\sum_{t=1}^{S}\sum_{i=1}^{C}\big|x^{(i)}_{m,L+t}-\hat{x}^{(i)}_{m,L+t}\big|
    \end{align*}

    \item \textbf{Hiệu năng}, đo theo cấu hình của TimeBase\refnote{Huang2025TimeBase}: Electricity ($C=321$, $L=S=720$):
    \begin{itemize}
        \item \textbf{Params}: số tham số; \textbf{MACs}: số phép nhân--cộng cho một mẫu.
        \item \textbf{Epoch Time}: thời gian một epoch huấn luyện; \textbf{Infer Time}: dự báo một mẫu trên CPU.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Nhận xét hiệu năng}
\begin{itemize}
    \item DyBAS-Net \textbf{nhanh thứ hai trong 16 mô hình} khi suy luận (1.63 ms một mẫu), chỉ sau TimeBase (1.01 ms).
    \item Nhanh hơn SparseTSF (2.39 ms), CycleNet/Linear (6.71 ms), FITS (7.48 ms), CycleNet/MLP (7.93 ms),
          TQNet (25.04 ms, gấp 15 lần) và GTR (28.37 ms, gấp 17 lần).
    \item So với các mô hình sâu: nhanh hơn iTransformer $\sim$29 lần, TimeMixer $\sim$500 lần, PatchTST $\sim$570 lần,
          MS-FLOW $\sim$1\,300 lần, TimesNet hơn 8\,000 lần.
    \item Huấn luyện nhẹ: 40 s một epoch.
\end{itemize}

          có điều chuẩn thưa (CrossVBlock, DropFusion, IntraVBlock của DyTimeNet), đặt trong \emph{không gian cơ sở}.
    \item \textbf{Rất nhẹ:} 23.8M MACs, 1.63 ms một mẫu trên CPU ở cấu hình Electricity; nhanh thứ hai trong 16 mô hình,
          nhanh hơn TQNet 15 lần và nhanh hơn iTransformer, TimeMixer, PatchTST, TimesNet từ 29 đến hơn 8\,000 lần.
    \item \textbf{Nhánh trộn biến có ích:} so với TimeBase cùng quy trình, giảm MSE trung bình trên cả 5/5 bộ dữ liệu (tới
          $3.7\%$ trên ETTm1).
\end{itemize}
